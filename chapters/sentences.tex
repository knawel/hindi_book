\subsection{3.1 Identity \& Description — \textit{honaa}}

\pattern{Subject + [Noun / Adjective] + honaa (conjugated)}

\begin{tabular}{@{}HRG@{}}
  \toprule
  Hindi & Transliteration & English \\
  \midrule
  \hw{मैं} student \hw{हूँ।}       & main student huun      & I am a student. \\

\pattern{Owner + kaa/kii/ke + Possessed noun}

\begin{tabular}{@{}HRG@{}}
  \toprule
  Hindi & Transliteration & English \\
  \midrule
  \hw{मेरा नाम जॉन है।}       & meraa naam John hai     & My name is John. \\
  \hw{उसकी किताब अच्छी है।}   & uskii kitaab achchhii hai  & Her book is good. \\
  \hw{हमारा घर बड़ा है।}       & hamaaraa ghar baRaa hai & Our house is big. \\
  \bottomrule
\end{tabular}

\subsection{3.13 From \ldots{} to \ldots{} — \textit{se \ldots{} tak}}

\pattern{[Place/Time A] + se + [Place/Time B] + tak}

\textit{se} marks the starting point, \textit{tak} the end point. Works for both places and times.
Clock time uses \textit{baje} (\hw{बजे}, ``o'clock''): \textit{subah 7 baje} = 7 in the morning.

\begin{tabular}{@{}HRG@{}}
  \toprule
  Hindi & Transliteration & English \\
  \midrule
  \hw{आज मैंने शक्ति से अघम तक साइकिल चलाई।} & aaj mainne Shakti se Agham tak saaikal chalaaii & Today I cycled from Shakti to Agham. \\
  \hw{सुबह 7 बजे से शाम 4 बजे तक}            & subah 7 baje se shaam 4 baje tak               & from 7 am to 4 pm \\
  \hw{लगभग 9 घंटे}                            & lagbhag 9 ghanTe                               & about 9 hours \\
  \hw{60 किलोमीटर, लेकिन सबसे ऊँची जगह 5400 मीटर पर थी।} & 60 kilomiiTar, lekin sabse uunchii jagah 5400 miiTar par thii & 60 km, but the highest point was at 5400 m. \\
  \bottomrule
\end{tabular}

\subsection{3.14 Experience — ``I have never \ldots'' (\textit{kabhii nahiin})}

\pattern{Subject\,+\,ne + kabhii (bhii) + Object + nahiin + verb-pp + hai/hain}

\textit{kabhii} = ever; \textit{kabhii nahiin} = never; \textit{kabhii bhii nahiin} = never ever (emphatic).
The verb agrees with the object: \textit{kuTTuu} (m.sg.) → \textit{khaayaa}; \textit{phuul} (m.pl.) → \textit{dekhe hain}.

\begin{tabular}{@{}HRG@{}}
  \toprule
  Hindi & Transliteration & English \\
  \midrule
  \hw{मैंने कभी कुट्टू नहीं खाया है।}          & mainne kabhii kuTTuu nahiin khaayaa hai     & I have never eaten buckwheat. \\
  \hw{मैंने कभी भी कुट्टू के फूल नहीं देखे हैं।} & mainne kabhii bhii kuTTuu ke phuul nahiin dekhe hain & I have never (ever) seen buckwheat flowers. \\
  \hw{क्या आपने कभी कुट्टू खाया है?}          & kyaa aapne kabhii kuTTuu khaayaa hai?      & Have you ever eaten buckwheat? \\
  \bottomrule
\end{tabular}

\subsection{3.15 Telling the Time — \textit{baje}}

\pattern{[number] + bajaa hai (1 o'clock) / baje hain (2+) \ \ \textnormal{— at:} [number] + baje}

\begin{tabular}{@{}HRG@{}}
  \toprule
  Hindi & Transliteration & English \\
  \midrule
  \hw{कितने बजे हैं?}          & kitne baje hain?        & What time is it? \\
  \hw{एक बजा है।}              & ek bajaa hai            & It's one o'clock. \\
  \hw{चार बजे हैं।}            & chaar baje hain         & It's four o'clock. \\
  \hw{सवा दो बजे हैं।}         & savaa do baje hain      & It's 2:15 (quarter past two). \\
  \hw{साढ़े तीन बजे हैं।}       & saaRhe tiin baje hain   & It's 3:30 (half past three). \\
  \hw{पौने पाँच बजे हैं।}       & paune paanch baje hain  & It's 4:45 (quarter to five). \\
  \hw{डेढ़ बजा है।}             & DeRh bajaa hai          & It's 1:30. \\
  \hw{ढाई बजे हैं।}             & Dhaaii baje hain        & It's 2:30. \\
  \hw{दो बजकर दस मिनट}          & do bajkar das minaT     & 2:10 (ten past two) \\
  \hw{तीन बजने में दस मिनट}      & tiin bajne men das minaT & 2:50 (ten to three) \\
  \hw{मैं सात बजे उठता हूँ।}    & main saat baje uThtaa huun & I get up at seven. \\
  \hw{क्लास शाम छह बजे है।}     & klaas shaam chhah baje hai & The class is at 6 pm. \\
  \bottomrule
\end{tabular}

\noindent\textit{Note:} \textit{saaRhe} is only used from 3:30 upwards; 1:30 and 2:30 have their own words, \textit{DeRh} and \textit{Dhaaii}.
